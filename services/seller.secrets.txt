# seller.secrets.txt — the secrets this repo's services need.
#
# `usvc_seller secrets upload seller.secrets.txt` seeds the comment-block
# description shown to customers (unitysvc/unitysvc#1618). Values are env-only
# — `${NAME}` with no default — so real keys must come from the environment: a
# GitHub Actions secret in CI, or your shell locally.
#
# Sourceable locally:  source ./seller.secrets.txt
#
# Every line is `export`ed on purpose. A plain `source` of a bare
# `NAME=value` file sets shell variables that are NOT passed to child
# processes, so `specs run-tests` would spawn its scripts without them and
# quietly fall back to the real providers.
#
# Every token here is a **customer** credential in normal use: the MCP channel
# carries `${ customer_secrets.<NAME> }`, so each customer brings their own
# token under the same name and the gateway injects it as
# `Authorization: Bearer <token>` on the upstream MCP call. The seller secret of
# the same name is documented here so customers know what to provide. Basic
# platform connectivity tests do not use fake tokens: they handshake with the
# real upstream endpoint and accept the provider's auth challenge as a live
# "registration required" response.
#
# Scope these down. Every token below reaches a real account, and an MCP client
# will call whatever tools it is offered — a broadly-scoped token is a standing
# invitation. Prefer read-only, single-project, short-lived wherever the
# provider allows it.
#
# ---------------------------------------------------------------------------
# The *_MCP_BASE variables below are endpoint overrides, not credentials. Each
# one defaults to the official upstream endpoint. Override only when a provider
# moves, when testing a staging provider endpoint, or when deliberately running
# a private deployment.
# ---------------------------------------------------------------------------

# GITHUB_PERSONAL_ACCESS_TOKEN — GitHub personal access token.
#   A GitHub PAT used as the bearer token for GitHub's hosted MCP server.
#   Prefer a fine-grained token limited to the specific repositories you want
#   the assistant to reach, with read-only permissions unless you actively want
#   it opening issues and pull requests. A classic token with `repo` scope works
#   but grants far more than most usage needs.
#
#   Create at GitHub → Settings → Developer settings → Personal access tokens.
export GITHUB_PERSONAL_ACCESS_TOKEN=${GITHUB_PERSONAL_ACCESS_TOKEN:-}

# LINEAR_API_KEY — Linear API key.
#   Your personal Linear API key, sent as the bearer token. It inherits your own
#   permissions across every team you belong to — Linear does not scope API keys
#   per team — so treat it as full account access and issue it from an account
#   whose reach you are comfortable exposing.
#
#   Create at Linear → Settings → Security & access → Personal API keys.
export LINEAR_API_KEY=${LINEAR_API_KEY:-}

# TAVILY_API_KEY — Tavily API key.
#   Your Tavily key (`tvly-…`), used as the bearer token for Tavily's hosted MCP
#   server. Tavily bills per search credit, so this key is a spend surface: an
#   assistant looping on searches consumes your quota. Consider a dedicated key
#   you can rotate independently of other Tavily usage.
#
#   Create at app.tavily.com → API keys.
export TAVILY_API_KEY=${TAVILY_API_KEY:-}

# POSTHOG_PERSONAL_API_KEY — PostHog personal API key.
#   A PostHog personal API key (`phx_…`), used as the bearer token. Scope it at
#   creation time: PostHog lets you pick both the projects it reaches and the
#   individual scopes it carries. Read scopes cover the useful analytics
#   questions; write scopes let an assistant edit dashboards, insights and
#   feature flags, which is rarely what you want first.
#
#   This is the *personal* API key, not the project API key that the PostHog
#   SDKs use for ingestion — the two are different credentials and the
#   ingestion one will not authenticate here.
#
#   Create at PostHog → Settings → Personal API keys.
export POSTHOG_PERSONAL_API_KEY=${POSTHOG_PERSONAL_API_KEY:-}

# SENTRY_AUTH_TOKEN — Sentry user auth token.
#   A Sentry auth token used as the bearer token. Grant the narrowest scopes
#   that answer your questions — `org:read`, `project:read` and `event:read`
#   cover issue triage; the write scopes let an assistant resolve and assign
#   issues. The token reaches every project in the organisations it is scoped
#   to.
#
#   Create at Sentry → Settings → Auth tokens (user tokens, not a DSN — a DSN is
#   an ingestion key and will not authenticate here).
export SENTRY_AUTH_TOKEN=${SENTRY_AUTH_TOKEN:-}

# GRAFANA_SERVICE_ACCOUNT_TOKEN — Grafana service account token.
#   A Grafana service account token (`glsa_…`) used as the bearer token. Create
#   a dedicated service account with the Viewer role rather than reusing an
#   admin one: the MCP tools can query datasources, and a datasource query is
#   only as safe as what that account is allowed to read.
#
#   Create at Grafana → Administration → Users and access → Service accounts.
export GRAFANA_SERVICE_ACCOUNT_TOKEN=${GRAFANA_SERVICE_ACCOUNT_TOKEN:-}

# ATLASSIAN_API_TOKEN — Atlassian API token.
#   An Atlassian Cloud API token, sent as the bearer token. It carries your own
#   permissions across every site you can reach — Atlassian tokens are not
#   scoped per project or per product — so issue it from an account whose
#   access you are comfortable exposing, and check your org's admin policy
#   before granting anything that can write.
#
#   Create at id.atlassian.com → Security → API tokens.
export ATLASSIAN_API_TOKEN=${ATLASSIAN_API_TOKEN:-}

# NOTION_API_TOKEN — Notion internal integration token.
#   The token of a Notion integration, sent as the bearer token. Notion scopes
#   access by *sharing* rather than by the token: an integration sees only the
#   pages and databases explicitly shared with it, so share one space first and
#   widen from there rather than granting workspace-wide access up front.
#
#   Create at notion.so/my-integrations.
export NOTION_API_TOKEN=${NOTION_API_TOKEN:-}

# SUPABASE_ACCESS_TOKEN — Supabase personal access token.
#   A Supabase personal access token, sent as the bearer token. It reaches every
#   organisation and project your account can, and these tools read schema and
#   data — issue a token you can revoke on its own, and prefer read-only usage.
#
#   Create at supabase.com/dashboard/account/tokens.
export SUPABASE_ACCESS_TOKEN=${SUPABASE_ACCESS_TOKEN:-}

# CLOUDFLARE_API_TOKEN — Cloudflare API token.
#   A Cloudflare API token, sent as the bearer token. Scope it narrowly:
#   Cloudflare tokens can be limited to specific accounts, zones and
#   permissions, and reading Workers observability needs far less than the
#   default templates grant.
#
#   Create at dash.cloudflare.com → My Profile → API Tokens.
export CLOUDFLARE_API_TOKEN=${CLOUDFLARE_API_TOKEN:-}

# STRIPE_API_KEY — Stripe restricted API key.
#   A Stripe **restricted** API key, sent as the bearer token. Scoping is not
#   optional here: a full secret key can move money, and an MCP client will call
#   whatever tool it is offered. Grant read permissions only unless you have a
#   specific reason not to, and use a test-mode key while evaluating.
#
#   Create at dashboard.stripe.com → Developers → API keys → Restricted keys.
export STRIPE_API_KEY=${STRIPE_API_KEY:-}
